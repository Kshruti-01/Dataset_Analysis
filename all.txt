st.markdown("""

class DatasetAnalyzer:
    """Calculates dataset architecture configurations safely"""
    def __init__(self, df: pd.DataFrame):
        self.df = df
        
    def get_basic_info(self) -> Dict[str, Any]:
        return {
            "rows": self.df.shape[0],
            "columns": self.df.shape[1],
            "missing_values": self.df.isnull().sum().sum(),
            "duplicates": self.df.duplicated().sum()
        }
    
    def get_column_info(self) -> pd.DataFrame:
        col_info = []
        for col in self.df.columns:
            sample_val = "NaN"
            if len(self.df[col].dropna()) > 0:
                sample_val = str(self.df[col].dropna().iloc[0])
            
            col_info.append({
                "Column Name": col,
                "Data Type": str(self.df[col].dtype),
                "Unique Values": self.df[col].nunique(),
                "Missing Values": self.df[col].isnull().sum(),
                "Sample Value": sample_val
            })
        return pd.DataFrame(col_info)


def load_dataset(uploaded_file) -> Tuple[pd.DataFrame, str]:
    """
    Scans the data matrix horizontally to find the row containing 
    the largest amount of columns, completely ignoring top metadata filters.
    """
    file_extension = uploaded_file.name.split('.')[-1].lower()
    
    if file_extension == 'csv':
        raw_data = uploaded_file.getvalue()
        result = chardet.detect(raw_data)
        encoding = result['encoding'] if result['encoding'] else 'utf-8'
        df_raw = pd.read_csv(io.BytesIO(raw_data), encoding=encoding, header=None)
    elif file_extension in ['xlsx', 'xls']:
        df_raw = pd.read_excel(uploaded_file, header=None)
    else:
        raise ValueError(f"Unsupported file type: .{file_extension}")
    
    if df_raw is None or df_raw.empty:
        return df_raw, uploaded_file.name

    # 🎯 Find the exact row containing the wide table header index 
    non_empty_counts = df_raw.notna().sum(axis=1).tolist()
    true_header_idx = non_empty_counts.index(max(non_empty_counts))
    
    # Slice the table cleanly starting directly at that index
    df_cleaned = df_raw.iloc[true_header_idx:].reset_index(drop=True)
    
    # Establish structural header mapping
    raw_headers = []
    for col_pos in range(df_cleaned.shape[1]):
        val = df_cleaned.iloc[0, col_pos]
        header_str = str(val).strip() if pd.notna(val) else ""
        
        if header_str.lower() in ['nan', 'none', ''] or header_str.startswith('Unnamed:'):
            raw_headers.append(f"Column_{col_pos + 1}")
        else:
            raw_headers.append(header_str)
            
    df_cleaned.columns = raw_headers
    df_cleaned = df_cleaned[1:].reset_index(drop=True)
    df_cleaned = df_cleaned.dropna(how='all')

    return df_cleaned, uploaded_file.name


def main():
    st.markdown('<div class="main-header">📊 Dataset Analyzer Platform</div>', unsafe_allow_html=True)
    st.markdown("---")
    
    with st.sidebar:
        st.markdown("## 📂 Upload Dataset")
        uploaded_file = st.file_uploader(
            "Choose a file (CSV or Excel)",
            type=['csv', 'xlsx', 'xls'],
            key="file_uploader_widget"
        )
        
        # Clear out state configurations cleanly when file is unmounted
        if uploaded_file is None:
            for k in ['current_df', 'current_file']:
                if k in st.session_state: 
                    del st.session_state[k]
                
    if uploaded_file is not None:
        try:
            # Check for fresh execution updates
            if 'current_df' not in st.session_state or st.session_state.current_file != uploaded_file.name:
                with st.spinner("Extracting comprehensive column matrix..."):
                    df, filename = load_dataset(uploaded_file)
                    st.session_state.current_df = df
                    st.session_state.current_file = filename
            
            df = st.session_state.current_df
            analyzer = DatasetAnalyzer(df)
            basic_info = analyzer.get_basic_info()
            
            # Display high-level metrics
            c1, c2, c3, c4 = st.columns(4)
            c1.markdown(f'<div class="metric-card"><h3>{basic_info["rows"]:,}</h3><p>Rows</p></div>', unsafe_allow_html=True)
            c2.markdown(f'<div class="metric-card"><h3>{basic_info["columns"]}</h3><p>Total Columns Detected</p></div>', unsafe_allow_html=True)
            c3.markdown(f'<div class="metric-card"><h3>{basic_info["missing_values"]:,}</h3><p>Missing Cells</p></div>', unsafe_allow_html=True)
            c4.markdown(f'<div class="metric-card"><h3>{basic_info["duplicates"]:,}</h3><p>Duplicate Rows</p></div>', unsafe_allow_html=True)
            
            st.markdown("---")
            
            tab1, tab2 = st.tabs(["📋 Dataset Preview", "🔍 Complete Column List"])
            
            with tab1:
                st.markdown('<div class="sub-header">Dataset Preview</div>', unsafe_allow_html=True)
                st.info("↔️ Use the horizontal scrollbar at the bottom of the table grid to browse all available layout columns.")
                
                # Render data with zero automated structural alterations or drop sequences
                st.dataframe(df.head(100), use_container_width=True)
                
            with tab2:
                st.markdown('<div class="sub-header">Complete Column Inventory Tracking</div>', unsafe_allow_html=True)
                col_info_df = analyzer.get_column_info()
                st.dataframe(col_info_df, use_container_width=True)
                    
        except Exception as e:
            st.error(f"Processing Failure: {str(e)}")
            st.info("Check whether your document has unclosed file stream hooks open locally inside Excel.")
    else:
        st.info("👈 Please select and upload your data document asset in the sidebar container.")


if __name__ == "__main__":
    main()
