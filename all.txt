class DatasetAnalyzer:
    """Main class for dataset analysis operations"""
    
    def __init__(self, df: pd.DataFrame, filename: str = ""):
        self.df = df
        self.filename = filename
        self.insights = []
        
    def get_basic_info(self) -> Dict[str, Any]:
        return {
            "rows": self.df.shape[0],
            "columns": self.df.shape[1],
            "memory_usage": self.df.memory_usage(deep=False).sum() / 1024**2,  # MB
            "duplicates": self.df.duplicated().sum(),
            "missing_values": self.df.isnull().sum().sum(),
            "missing_percentage": (self.df.isnull().sum().sum() / (self.df.shape[0] * self.df.shape[1])) * 100 if self.df.shape[0] > 0 else 0
        }
    
    def get_column_info(self) -> pd.DataFrame:
        col_info = []
        for col in self.df.columns:
            sample_val = "NaN"
            if len(self.df[col].dropna()) > 0:
                sample_val = str(self.df[col].dropna().iloc[0])
            
            col_info.append({
                "Column": col,
                "Type": str(self.df[col].dtype),
                "Unique Values": self.df[col].nunique(),
                "Missing Values": self.df[col].isnull().sum(),
                "Missing %": round((self.df[col].isnull().sum() / len(self.df)) * 100, 2) if len(self.df) > 0 else 0,
                "Sample Value": sample_val
            })
        return pd.DataFrame(col_info)
    
    def get_numeric_stats(self) -> pd.DataFrame:
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns
        if len(numeric_cols) > 0:
            return self.df[numeric_cols].describe().T
        return pd.DataFrame()
    
    def detect_schema(self) -> Dict[str, Any]:
        schema = {
            "column_count": len(self.df.columns),
            "row_count": len(self.df),
            "columns": {}
        }
        for col in self.df.columns:
            dtype = self.df[col].dtype
            unique_count = self.df[col].nunique()
            missing_count = self.df[col].isnull().sum()
            
            if dtype in ['int64', 'float64']:
                col_type = "numeric"
            elif dtype == 'object':
                if len(self.df) > 0 and (unique_count / len(self.df) < 0.05):
                    col_type = "categorical"
                else:
                    col_type = "text"
            elif 'datetime' in str(dtype):
                col_type = "datetime"
            else:
                col_type = "other"
            
            raw_samples = self.df[col].dropna().head(3).tolist()
            safe_samples = []
            for val in raw_samples:
                if isinstance(val, (np.integer, np.int64)):
                    safe_samples.append(int(val))
                elif isinstance(val, (np.floating, np.float64)):
                    safe_samples.append(float(val))
                else:
                    safe_samples.append(str(val))
            
            schema["columns"][col] = {
                "data_type": str(dtype),
                "unique_values": int(unique_count),
                "missing_values": int(missing_count),
                "column_type": col_type,
                "sample_values": safe_samples
            }
        return schema
    
    def generate_insights(self) -> List[str]:
        insights = []
        basic_info = self.get_basic_info()
        insights.append(f"📊 Dataset contains **{basic_info['rows']:,}** rows and **{basic_info['columns']}** columns.")
        
        if basic_info['missing_values'] > 0:
            insights.append(f"⚠️ Found **{basic_info['missing_values']:,}** missing values ({basic_info['missing_percentage']:.1f}% of total data).")
        
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns.tolist()
        categorical_cols = self.df.select_dtypes(include=['object']).columns.tolist()
        insights.append(f"📝 Contains **{len(numeric_cols)}** numeric columns and **{len(categorical_cols)}** categorical columns.")
        
        return insights

    def create_visualizations(self) -> Dict[str, Any]:
        plots = {}
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns.tolist()
        if len(numeric_cols) >= 2:
            corr_matrix = self.df[numeric_cols].corr()
            fig = px.imshow(
                corr_matrix, text_auto=True, aspect="auto",
                title="Correlation Heatmap", color_continuous_scale="RdBu_r"
            )
            plots["correlation"] = fig
        return plots


class DataCleaner:
    """Handles data cleaning operations"""
    @staticmethod
    def remove_duplicates(df: pd.DataFrame) -> pd.DataFrame:
        return df.drop_duplicates()
    
    @staticmethod
    def handle_missing_values(df: pd.DataFrame, strategy: str = "drop") -> pd.DataFrame:
        if strategy == "drop":
            return df.dropna()
        return df


def load_dataset(uploaded_file) -> Tuple[pd.DataFrame, str]:
    """
    Smart layout scanning matrix parser. Locates the widest row to capture
    every single horizontal structural column block (A to CY).
    """
    file_extension = uploaded_file.name.split('.')[-1].lower()
    
    if file_extension == 'csv':
        raw_data = uploaded_file.getvalue()
        result = chardet.detect(raw_data)
        encoding = result['encoding'] if result['encoding'] else 'utf-8'
        df_raw = pd.read_csv(io.BytesIO(raw_data), encoding=encoding, header=None)
    elif file_extension in ['xlsx', 'xls']:
        df_raw = pd.read_excel(uploaded_file, header=None)
    else:
        raise ValueError(f"Unsupported file type: .{file_extension}")
    
    if df_raw is None or df_raw.empty:
        return df_raw, uploaded_file.name

    # 🎯 SCANNING AXIS: Find the row that holds the actual complete horizontal table headers
    non_empty_counts = df_raw.notna().sum(axis=1).tolist()
    true_header_idx = non_empty_counts.index(max(non_empty_counts))
    
    df_cleaned = df_raw.iloc[true_header_idx:].reset_index(drop=True)
    
    # Extract headers cleanly across all available dimensions
    raw_headers = []
    for col_pos in range(df_cleaned.shape[1]):
        val = df_cleaned.iloc[0, col_pos]
        header_str = str(val).strip() if pd.notna(val) else ""
        
        if header_str.lower() in ['nan', 'none', ''] or header_str.startswith('Unnamed:'):
            raw_headers.append(f"Column_{col_pos + 1}")
        else:
            raw_headers.append(header_str)
            
    df_cleaned.columns = raw_headers
    df_cleaned = df_cleaned[1:].reset_index(drop=True)
    df_cleaned = df_cleaned.dropna(how='all')

    return df_cleaned, uploaded_file.name
