class DatasetAnalyzer:
    """Main class for dataset analysis operations"""
    
    def __init__(self, df: pd.DataFrame, filename: str = ""):
        self.df = df
        self.filename = filename
        self.insights = []
        
    def get_basic_info(self) -> Dict[str, Any]:
        """Get basic dataset information safely and quickly"""
        return {
            "rows": self.df.shape[0],
            "columns": self.df.shape[1],
            "memory_usage": self.df.memory_usage(deep=False).sum() / 1024**2,  # MB
            "duplicates": self.df.duplicated().sum(),
            "missing_values": self.df.isnull().sum().sum(),
            "missing_percentage": (self.df.isnull().sum().sum() / (self.df.shape[0] * self.df.shape[1])) * 100 if self.df.shape[0] > 0 else 0
        }
    
    def get_column_info(self) -> pd.DataFrame:
        """Get detailed column information"""
        col_info = []
        for col in self.df.columns:
            col_info.append({
                "Column": col,
                "Type": str(self.df[col].dtype),
                "Unique Values": self.df[col].nunique(),
                "Missing Values": self.df[col].isnull().sum(),
                "Missing %": round((self.df[col].isnull().sum() / len(self.df)) * 100, 2) if len(self.df) > 0 else 0,
                "Sample Value": str(self.df[col].dropna().iloc[0]) if len(self.df[col].dropna()) > 0 else "NaN"
            })
        
        return pd.DataFrame(col_info)
    
    def get_numeric_stats(self) -> pd.DataFrame:
        """Get statistics for numeric columns"""
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns
        if len(numeric_cols) > 0:
            return self.df[numeric_cols].describe().T
        return pd.DataFrame()
    
    def detect_schema(self) -> Dict[str, Any]:
        """Detect and return schema information safely serialized for Streamlit JSON view"""
        schema = {
            "column_count": len(self.df.columns),
            "row_count": len(self.df),
            "columns": {}
        }
        
        for col in self.df.columns:
            dtype = self.df[col].dtype
            unique_count = self.df[col].nunique()
            missing_count = self.df[col].isnull().sum()
            
            # Determine column role
            if dtype in ['int64', 'float64']:
                col_type = "numeric"
            elif dtype == 'object':
                if len(self.df) > 0 and (unique_count / len(self.df) < 0.05):
                    col_type = "categorical"
                else:
                    col_type = "text"
            elif 'datetime' in str(dtype):
                col_type = "datetime"
            else:
                col_type = "other"
            
            # Sanitize sample values explicitly to standard native Python classes
            raw_samples = self.df[col].dropna().head(3).tolist()
            safe_samples = []
            for val in raw_samples:
                if isinstance(val, (np.integer, np.int64)):
                    safe_samples.append(int(val))
                elif isinstance(val, (np.floating, np.float64)):
                    safe_samples.append(float(val))
                elif pd.api.types.is_datetime64_any_dtype(type(val)) or hasattr(val, 'isoformat'):
                    safe_samples.append(str(val))
                else:
                    safe_samples.append(val)
            
            schema["columns"][col] = {
                "data_type": str(dtype),
                "unique_values": int(unique_count),
                "missing_values": int(missing_count),
                "column_type": col_type,
                "sample_values": safe_samples
            }
        
        return schema
    
    def generate_insights(self) -> List[str]:
        """Generate automatic insights from the dataset"""
        insights = []
        basic_info = self.get_basic_info()
        
        insights.append(f"Dataset contains **{basic_info['rows']:,}** rows and **{basic_info['columns']}** columns.")
        
        if basic_info['missing_values'] > 0:
            insights.append(f"Found **{basic_info['missing_values']:,}** missing values ({basic_info['missing_percentage']:.1f}% of total data).")
        else:
            insights.append("No missing values found in the dataset.")
        
        if basic_info['duplicates'] > 0:
            insights.append(f"Detected **{basic_info['duplicates']:,}** duplicate rows.")
        else:
            insights.append("No duplicate rows found.")
        
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns.tolist()
        categorical_cols = self.df.select_dtypes(include=['object']).columns.tolist()
        
        insights.append(f"📝 Contains **{len(numeric_cols)}** numeric columns and **{len(categorical_cols)}** categorical columns.")
        
        for col in numeric_cols[:3]:
            stats = self.df[col].describe()
            if not stats.empty and pd.notnull(stats['min']):
                insights.append(f"Column '{col}': range [{stats['min']:.2f}, {stats['max']:.2f}], mean={stats['mean']:.2f}")
        
        for col in categorical_cols[:3]:
            if len(self.df) > 0:
                unique_pct = (self.df[col].nunique() / len(self.df)) * 100
                if unique_pct > 50 and self.df[col].nunique() > 50:
                    insights.append(f"Column '{col}' has high cardinality ({self.df[col].nunique():,} unique values).")
        
        if len(numeric_cols) >= 2:
            corr_matrix = self.df[numeric_cols].corr()
            for i in range(len(corr_matrix.columns)):
                for j in range(i+1, len(corr_matrix.columns)):
                    corr_val = corr_matrix.iloc[i, j]
                    if abs(corr_val) > 0.7:
                        insights.append(f"Strong correlation ({corr_val:.2f}) between '{corr_matrix.columns[i]}' and '{corr_matrix.columns[j]}'.")
        
        return insights
    
    def create_visualizations(self) -> Dict[str, Any]:
        """Create structured visualizations filtered against edge-case high-cardinality values"""
        plots = {}
        numeric_cols = self.df.select_dtypes(include=[np.number]).columns.tolist()
        categorical_cols = self.df.select_dtypes(include=['object']).columns.tolist()
        
        # 1. Distribution plots
        if numeric_cols:
            for col in numeric_cols[:3]:
                fig = px.histogram(
                    self.df, x=col, nbins=50, 
                    title=f"Distribution of {col}",
                    labels={col: col, "count": "Frequency"},
                    color_discrete_sequence=["#1E88E5"]
                )
                fig.update_layout(height=400)
                plots[f"dist_{col}"] = fig
        
        # 2. Box plots
        if len(numeric_cols) >= 2:
            numeric_subset = self.df[numeric_cols[:5]]
            melted = numeric_subset.melt(var_name="Variable", value_name="Value")
            fig = px.box(
                melted, x="Variable", y="Value",
                title="Box Plots for Numeric Columns",
                color="Variable"
            )
            fig.update_layout(height=450)
            plots["boxplots"] = fig
        
        # 3. Correlation Heatmap
        if len(numeric_cols) >= 2:
            corr_matrix = self.df[numeric_cols].corr()
            fig = px.imshow(
                corr_matrix, 
                text_auto=True, 
                aspect="auto",
                title="Correlation Heatmap",
                color_continuous_scale="RdBu_r",
                zmin=-1, zmax=1
            )
            fig.update_layout(height=500)
            plots["correlation"] = fig
        
        # 4. Filtered Categorical Bar Charts
        if categorical_cols:
            visualized_count = 0
            for col in categorical_cols:
                if visualized_count >= 2:
                    break
                
                if len(self.df) > 0:
                    unique_pct = self.df[col].nunique() / len(self.df)
                    if unique_pct > 0.5 and self.df[col].nunique() > 50:
                        continue  # Bypass true unique string identifiers or hashes
                
                value_counts = self.df[col].value_counts().head(15)
                if value_counts.empty:
                    continue
                    
                fig = px.bar(
                    x=value_counts.values, 
                    y=value_counts.index.astype(str),
                    orientation='h',
                    title=f"Top Categories in {col}",
                    labels={"x": "Count", "y": col},
                    color=value_counts.values,
                    color_continuous_scale="Blues"
                )
                fig.update_layout(height=400)
                plots[f"cat_{col}"] = fig
                visualized_count += 1
        
        # 5. Missing values heatmap
        if self.df.isnull().sum().sum() > 0:
            viz_df = self.df
            is_sampled = False
            if len(self.df) > 5000:
                viz_df = self.df.sample(n=5000, random_state=42).sort_index()
                is_sampled = True
                
            missing_data = viz_df.isnull().astype(int)
            title_suffix = " (Sampled max 5,000 rows)" if is_sampled else ""
            fig = px.imshow(
                missing_data.T,
                title=f"Missing Values Heatmap{title_suffix}",
                labels={"x": "Row Index", "y": "Column"},
                color_continuous_scale=["#2ecc71", "#e74c3c"],
                aspect="auto"
            )
            fig.update_layout(height=max(300, len(self.df.columns) * 20))
            plots["missing"] = fig
        
        return plots


class DataCleaner:
    """Handles data cleaning operations with safe type conversion overrides"""
    
    @staticmethod
    def remove_duplicates(df: pd.DataFrame) -> pd.DataFrame:
        return df.drop_duplicates()
    
    @staticmethod
    def handle_missing_values(df: pd.DataFrame, strategy: str = "drop", columns: List[str] = None) -> pd.DataFrame:
        df_clean = df.copy()
        if columns is None:
            cols_to_handle = df.columns
        else:
            cols_to_handle = [c for c in columns if c in df.columns]
        
        if strategy == "drop":
            df_clean = df_clean.dropna(subset=cols_to_handle)
        elif strategy == "fill_mean":
            for col in cols_to_handle:
                if df_clean[col].dtype in ['int64', 'float64']:
                    df_clean[col] = df_clean[col].fillna(df_clean[col].mean())
        elif strategy == "fill_median":
            for col in cols_to_handle:
                if df_clean[col].dtype in ['int64', 'float64']:
                    df_clean[col] = df_clean[col].fillna(df_clean[col].median())
        elif strategy == "fill_mode":
            for col in cols_to_handle:
                mode_series = df_clean[col].mode()
                if not mode_series.empty:
                    df_clean[col] = df_clean[col].fillna(mode_series[0])
                else:
                    if df_clean[col].dtype in ['int64', 'float64']:
                        df_clean[col] = df_clean[col].fillna(0)
                    else:
                        df_clean[col] = df_clean[col].fillna("Unknown")
        elif strategy == "fill_constant":
            for col in cols_to_handle:
                if df_clean[col].dtype in ['int64', 'float64']:
                    df_clean[col] = df_clean[col].fillna(0)
                else:
                    df_clean[col] = df_clean[col].fillna("Fixed")
        
        return df_clean
    
    @staticmethod
    def convert_data_types(df: pd.DataFrame) -> pd.DataFrame:
        """Attempt to convert columns to native numerical or datetime parameters"""
        df_clean = df.copy()
        for col in df_clean.columns:
            if df_clean[col].dtype == 'object':
                # Try to convert to numeric
                try:
                    df_clean[col] = pd.to_numeric(df_clean[col])
                    continue
                except:
                    pass
                
                # Try to convert to datetime
                try:
                    sample_str = str(df_clean[col].dropna().iloc[0]) if df_clean[col].dropna().shape[0] > 0 else ""
                    if re.search(r'\d{2,4}[-/]\d{2}[-/]\d{2,4}', sample_str) or 'Feb' in sample_str or 'Jan' in sample_str:
                        df_clean[col] = pd.to_datetime(df_clean[col])
                except:
                    pass
        return df_clean
    
    @staticmethod
    def remove_outliers(df: pd.DataFrame, columns: List[str] = None, method: str = "iqr") -> pd.DataFrame:
        """Applies holistic boolean masks to eliminate cascading table data wiping loops"""
        df_clean = df.copy()
        if columns is None:
            columns = df.select_dtypes(include=[np.number]).columns.tolist()
        
        if not columns:
            return df_clean
            
        mask = pd.Series(True, index=df_clean.index)
        
        for col in columns:
            if col in df_clean.columns and df_clean[col].dtype in ['int64', 'float64']:
                if df_clean[col].nunique() <= 1:
                    continue
                    
                if method == "iqr":
                    Q1 = df_clean[col].quantile(0.25)
                    Q3 = df_clean[col].quantile(0.75)
                    IQR = Q3 - Q1
                    lower_bound = Q1 - 1.5 * IQR
                    upper_bound = Q3 + 1.5 * IQR
                    mask = mask & (df_clean[col] >= lower_bound) & (df_clean[col] <= upper_bound)
                elif method == "zscore":
                    mean = df_clean[col].mean()
                    std = df_clean[col].std()
                    if std > 0:
                        mask = mask & (abs(df_clean[col] - mean) <= 3 * std)
        
        if mask.sum() == 0:
            return df_clean  # Safety switch fallback to prevent deleting data completely
        return df_clean[mask]


def load_dataset(uploaded_file) -> Tuple[pd.DataFrame, str]:
    """
    Advanced automated handler for loading files. Automatically flags and corrects 
    leading metadata spacing blocks, standard rows, and transposed data key configurations.
    Only supports CSV and Excel extensions.
    """
    file_extension = uploaded_file.name.split('.')[-1].lower()
    df_raw = None
    
    if file_extension == 'csv':
        raw_data = uploaded_file.getvalue()
        result = chardet.detect(raw_data)
        encoding = result['encoding'] if result['encoding'] else 'utf-8'
        try:
            df_raw = pd.read_csv(io.BytesIO(raw_data), encoding=encoding, header=None)
        except:
            df_raw = pd.read_csv(io.BytesIO(raw_data), encoding='utf-8', engine='python', header=None)
        
    elif file_extension in ['xlsx', 'xls']:
        df_raw = pd.read_excel(uploaded_file, header=None)
    else:
        raise ValueError(f"Unsupported file type: .{file_extension}. Please upload a CSV or Excel file.")
    
    if df_raw is None or df_raw.empty:
        return df_raw, uploaded_file.name

    # Header Location Scanner Loop
    header_row_idx = 0
    for idx, row in df_raw.iterrows():
        valid_cells = row.dropna().tolist()
        if len(valid_cells) > 0:
            string_cells = [str(c) for c in valid_cells if not str(c).strip().replace('.', '', 1).isdigit()]
            if len(string_cells) >= len(valid_cells) * 0.5 or idx == df_raw.dropna(how='all').index[0]:
                header_row_idx = idx
                break

    df_cleaned = df_raw.iloc[header_row_idx:].reset_index(drop=True)
    df_cleaned.columns = [str(c).strip() for c in df_cleaned.iloc[0]]
    df_cleaned = df_cleaned[1:].reset_index(drop=True)
    
    df_cleaned = df_cleaned.loc[:, df_cleaned.columns.notna() & (df_cleaned.columns != 'None') & (df_cleaned.columns != 'nan')]
    df_cleaned = df_cleaned.loc[:, ~df_cleaned.columns.str.startswith('Unnamed:')]

    if df_cleaned.empty:
        return df_cleaned, uploaded_file.name

    # Transposition Evaluator
    if df_cleaned.shape[1] == 2:
        first_col_values = df_cleaned.iloc[:, 0].dropna().tolist()
        unique_keys_count = len(set(first_col_values))
        
        if unique_keys_count == len(first_col_values) and len(first_col_values) > 1:
            df_transposed = df_cleaned.set_index(df_cleaned.columns[0]).T
            df_transposed.columns.name = None
            df_transposed = df_transposed.reset_index(drop=True)
            df_transposed.columns = [str(c).strip() for c in df_transposed.columns]
            return df_transposed, uploaded_file.name

    return df_cleaned, uploaded_file.name

                            )
                        elif export_format == "Excel":
                            output = io.BytesIO()
                            with pd.ExcelWriter(output, engine='openpyxl') as writer:
                                df.to_excel(writer, index=False, sheet_name='Cleaned_Data')
                            st.download_button(
                                label="Download Excel",
                                data=output.getvalue(),
                                file_name=f"cleaned_dataset_{st.session_state.current_file.split('.')[0]}.xlsx",
                                mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
                            )
                
                st.markdown("---")
                st.markdown("Reset to Original")
                if st.button("Reset to Original Dataset"):
                    st.session_state.current_df = st.session_state.original_df.copy()
                    st.success("Dataset reset to original structure settings!")
                    st.rerun()
        
        except Exception as e:
            st.error(f"Error loading dataset: {str(e)}")
            st.info("Ensure your file follows clean table standard configurations.")
    else:
        st.info("upload a dataset file (CSV or Excel) to begin analysis.")


if __name__ == "__main__":
    main()
